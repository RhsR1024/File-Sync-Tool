New Module: local_exec.rs

local_exec.rs — Public API
/// Execute local script groups for a completed copy task
pub fn execute_local_scripts(
    app_handle: &AppHandle,
    binding: &LocalScriptBinding,
    all_groups: &[LocalCommandGroup],
    local_folder_path: &Path,
    folder_name: &str,
    source_path: &str,
    should_cancel: Arc<AtomicBool>,
) -> LocalExecResult

/// Resolve a single command: auto-detect interpreter or pass through
fn resolve_command(raw: &str, vars: &VarMap) -> ResolvedCommand

/// Run one command as subprocess, capture output, emit logs
fn run_single_command(
    cmd: &ResolvedCommand,
    working_dir: &Path,
    app_handle: &AppHandle,

// Return type
struct LocalExecResult {
    success: bool,
    group_results: Vec<GroupResult>,  // per-group success/failure
    aborted: bool,                    // true if on_failure=abort triggered
}

scanner.rs Changes (Orchestration)

scanner.rs — Post-Copy Orchestration (pseudocode)
// After copy completes, read live config
let has_local = local_script_binding.is_some();
let has_remote = deploy_enabled && !server_bindings.is_empty();

match (has_local, has_remote, execution_order) {
    (true, true, LocalFirst) => {
        let local_result = execute_local_scripts(...);
        if local_result.aborted {
            emit_log("Local script aborted, skipping deploy");
            return;
        }
        deploy_to_remote(...);
    }
    (true, true, RemoteFirst) => {
        deploy_to_remote(...);
        execute_local_scripts(...);
    }
    (true, true, Parallel) => {
        // spawn_blocking for local, existing blocking for remote
        let (local_r, remote_r) = join!(local_task, remote_task);
    }
    (true, false, _) => execute_local_scripts(...),
    (false, true, _) => deploy_to_remote(...),
    (false, false, _) => { /* nothing to do */ }
}

on_failure Abort Behavior

Abort Scope

Within local scripts: If a group with on_failure: abort fails, skip remaining local groups.

Cross to remote deploy: Only blocks remote deploy when execution_order = local_first. In parallel mode, remote deploy is already running and won't be stopped.

In remote_first: Remote deploy failure does NOT block local scripts (they are independent concerns).
